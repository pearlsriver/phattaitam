type Props = { params: Promise<{ slug: string }> };

/** Supabase có thể trả quan hệ dạng object hoặc mảng — lấy phần tử đầu */
function one<T>(v: unknown): T | null {
  if (Array.isArray(v)) return (v[0] as T) ?? null;
  return (v as T) ?? null;
}

/** Thẻ SEO + ảnh/tiêu đề khi chia sẻ link qua Facebook, Zalo… */
export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params;
  const { data: post } = await serverSupabase()
    .from("posts")
    .select("title, excerpt, content, cover_image, tags, author_name, created_at, updated_at, author:profiles(full_name)")
    .eq("slug", decodeURIComponent(slug))
    .eq("is_published", true)
    .maybeSingle();

  if (!post) return { title: "Không tìm thấy bài viết" };

  const tags: string[] = Array.isArray(post.tags) ? post.tags : [];
  const description = post.excerpt || plainExcerpt(post.content, 180) || "Bài viết trên Phật Tại Tâm.";
  const image = post.cover_image || firstImage(post.content);
  const author = post.author_name || one<{ full_name: string | null }>(post.author)?.full_name || undefined;
  const url = `/bai-viet/${slug}`;

  return {
    title: post.title,
    description,
    keywords: tags,
    alternates: { canonical: url },
    authors: author ? [{ name: author }] : undefined,
    openGraph: {
      type: "article",
      url,
      title: post.title,
      description,
      publishedTime: post.created_at,
      modifiedTime: post.updated_at,
      authors: author ? [author] : undefined,
      tags,
      images: [{ url: image || DEFAULT_OG_IMAGE }],
    },
    twitter: { card: "summary_large_image", title: post.title, description, images: [image || DEFAULT_OG_IMAGE] },
  };
}

import type { Song } from "@/types/song";

export type RepeatMode = "off" | "all" | "one";

interface PlayerState {
  current: Song | null;
  queue: Song[];
  index: number;
  radio: boolean;
  /** Tên nguồn đang phát: "Radio", tên album, tên playlist… */
  source: string | null;
  playing: boolean;
  currentTime: number;
  duration: number;
  volume: number;
  muted: boolean;
  repeat: RepeatMode;
  playSong: (song: Song, list?: Song[], source?: string) => void;
  playList: (list: Song[], source?: string) => void;
  startRadio: (list: Song[], source?: string, shuffle?: boolean) => void;
  togglePlay: () => void;
  next: () => void;
  prev: () => void;
  jumpTo: (index: number) => void;
  seek: (seconds: number) => void;
  setVolume: (v: number) => void;
  toggleMute: () => void;
  cycleRepeat: () => void;
  stop: () => void;
}

const PlayerContext = createContext<PlayerState | null>(null);

function shuffle<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

export function PlayerProvider({ children }: { children: React.ReactNode }) {
  const audioRef = useRef<HTMLAudioElement>(null);
  const [queue, setQueue] = useState<Song[]>([]);
  const [index, setIndex] = useState(0);
  const [radio, setRadio] = useState(false);
  const [radioShuffle, setRadioShuffle] = useState(true);
  const [source, setSource] = useState<string | null>(null);
  const [token, setToken] = useState(0);
  const [playing, setPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);
  const [volume, setVolumeState] = useState(0.9);
  const [muted, setMuted] = useState(false);
  const [repeat, setRepeat] = useState<RepeatMode>("off");

  const bump = () => setToken((t) => t + 1);
  const current = queue[index] ?? null;

  /* Nhớ âm lượng & chế độ lặp */
  useEffect(() => {
    try {
      const v = parseFloat(localStorage.getItem("player:volume") ?? "");
      if (!Number.isNaN(v)) setVolumeState(Math.min(1, Math.max(0, v)));
      const r = localStorage.getItem("player:repeat");
      if (r === "all" || r === "one" || r === "off") setRepeat(r);
    } catch {}
  }, []);

  useEffect(() => {
    const a = audioRef.current;
    if (a) {
      a.volume = volume;
      a.muted = muted;
    }
  }, [volume, muted]);

  /* Bắt đầu phát khi đổi bài */
  useEffect(() => {
    const a = audioRef.current;
    if (!a || !current) return;
    a.src = songAudioUrl(current);
    a.currentTime = 0;
    setCurrentTime(0);
    setDuration(0);
    a.play().catch(() => setPlaying(false));
    countSongPlay(current.id).catch(() => {});
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [current?.id, token]);

  const playSong = useCallback((song: Song, list?: Song[], src?: string) => {
    const q = list && list.length ? list : [song];
    setQueue(q);
    setIndex(Math.max(0, q.findIndex((s) => s.id === song.id)));
    setRadio(false);
    setSource(src ?? null);
    bump();
  }, []);

  const playList = useCallback((list: Song[], src?: string) => {
    if (!list.length) return;
    setQueue(list);
    setIndex(0);
    setRadio(false);
    setSource(src ?? null);
    bump();
  }, []);

  /** Radio: phát danh sách (ngẫu nhiên hoặc theo thứ tự), hết vòng tự phát lại từ đầu */
  const startRadio = useCallback((list: Song[], src?: string, shuffleOn = true) => {
    if (!list.length) return;
    setQueue(shuffleOn ? shuffle(list) : list);
    setIndex(0);
    setRadio(true);
    setRadioShuffle(shuffleOn);
    setSource(src ?? "Radio Phật Tại Tâm");
    bump();
  }, []);

  const next = useCallback(() => {
    if (index + 1 < queue.length) {
      setIndex(index + 1);
      bump();
    } else if (radio) {
      setQueue(radioShuffle ? shuffle(queue) : queue);
      setIndex(0);
      bump();
    } else if (repeat === "all" && queue.length) {
      setIndex(0);
      bump();
    }
  }, [index, queue, radio, radioShuffle, repeat]);

  const prev = useCallback(() => {
    const a = audioRef.current;
    if (a && a.currentTime > 3) {
      a.currentTime = 0;
      return;
    }
    if (index > 0) {
      setIndex(index - 1);
      bump();
    } else if (a) {
      a.currentTime = 0;
    }
  }, [index]);

  const jumpTo = useCallback((i: number) => {
    setIndex(i);
    bump();
  }, []);

  const togglePlay = useCallback(() => {
    const a = audioRef.current;
    if (!a || !current) return;
    if (a.paused) a.play().catch(() => {});
    else a.pause();
  }, [current]);

  const seek = useCallback((s: number) => {
    const a = audioRef.current;
    if (a && Number.isFinite(s)) {
      a.currentTime = s;
      setCurrentTime(s);
    }
  }, []);

  const setVolume = useCallback((v: number) => {
    const x = Math.min(1, Math.max(0, v));
    setVolumeState(x);
    if (x > 0) setMuted(false);
    try { localStorage.setItem("player:volume", String(x)); } catch {}
  }, []);

  const toggleMute = useCallback(() => setMuted((m) => !m), []);

  const cycleRepeat = useCallback(() => {
    setRepeat((r) => {
      const n: RepeatMode = r === "off" ? "all" : r === "all" ? "one" : "off";
      try { localStorage.setItem("player:repeat", n); } catch {}
      return n;
    });
  }, []);

  const stop = useCallback(() => {
    audioRef.current?.pause();
    setQueue([]);
    setIndex(0);
    setRadio(false);
    setSource(null);
    setPlaying(false);
  }, []);

  /* Hết bài: lặp một bài / bài kế tiếp / lặp lại danh sách / dừng */
  const onEnded = useCallback(() => {
    const a = audioRef.current;
    if (repeat === "one" && a) {
      a.currentTime = 0;
      a.play().catch(() => {});
      return;
    }
    if (index + 1 < queue.length || radio || (repeat === "all" && queue.length)) next();
    else setPlaying(false);
  }, [repeat, index, queue.length, radio, next]);

  /* Điều khiển từ màn hình khoá / tai nghe */
  useEffect(() => {
    if (!current || typeof navigator === "undefined" || !("mediaSession" in navigator)) return;
    const ms = navigator.mediaSession;
    ms.metadata = new MediaMetadata({
      title: current.title,
      artist: current.artist ?? "Phật Tại Tâm",
      album: current.album_info?.title ?? source ?? "",
    });
    ms.setActionHandler("play", () => audioRef.current?.play().catch(() => {}));
    ms.setActionHandler("pause", () => audioRef.current?.pause());
    ms.setActionHandler("nexttrack", next);
    ms.setActionHandler("previoustrack", prev);
    ms.setActionHandler("seekto", (d) => { if (typeof d.seekTime === "number") seek(d.seekTime); });
  }, [current, source, next, prev, seek]);

  return (
    <PlayerContext.Provider
      value={{
        current, queue, index, radio, source, playing, currentTime, duration, volume, muted, repeat,
        playSong, playList, startRadio, togglePlay, next, prev, jumpTo, seek, setVolume, toggleMute, cycleRepeat, stop,
      }}
    >
      {children}
      <audio
        ref={audioRef}
        preload="auto"
        onPlay={() => setPlaying(true)}
        onPause={() => setPlaying(false)}
        onTimeUpdate={(e) => setCurrentTime(e.currentTarget.currentTime)}
        onLoadedMetadata={(e) => setDuration(e.currentTarget.duration)}
        onDurationChange={(e) => setDuration(e.currentTarget.duration)}
        onEnded={onEnded}
        onError={() => setPlaying(false)}
      />
    </PlayerContext.Provider>
  );
}

export function usePlayer() {
  const ctx = useContext(PlayerContext);
  if (!ctx) throw new Error("usePlayer phải dùng bên trong PlayerProvider");

async function accessToken(): Promise<string> {
  const { data } = await supabase.auth.getSession();
  const token = data.session?.access_token;
  if (!token) throw new Error("Phiên đăng nhập đã hết hạn, vui lòng đăng nhập lại.");
  return token;
}

/**
 * Tải tệp lên kho lưu trữ và trả về đường dẫn "<userId>/<thời gian>-<tên tệp>".
 * - Có NEXT_PUBLIC_MEDIA_URL: tải thẳng lên Cloudflare R2 (qua URL đã ký).
 * - Chưa có: dùng Supabase Storage như trước.
 */
export async function uploadFile(bucket: Bucket, file: File, userId: string, prefix = ""): Promise<string> {
  if (!MEDIA_URL) {
    const path = `${userId}/${prefix}${Date.now()}-${safeFileName(file.name)}`;
    const { error } = await supabase.storage.from(bucket).upload(path, file, { contentType: file.type || undefined });
    if (error) throw error;
    return path;
  }

  const res = await fetch("/api/media/upload-url", {
    method: "POST",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${await accessToken()}` },
    body: JSON.stringify({ bucket, filename: file.name, size: file.size, prefix }),
  });
  const info = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(info.error || "Không xin được quyền tải lên.");

  let put: Response;
  try {
    put = await fetch(info.uploadUrl, { method: "PUT", headers: { "Content-Type": info.contentType }, body: file });
  } catch {
    throw new Error("Không kết nối được Cloudflare R2. Kiểm tra cấu hình CORS của bucket.");
  }
  if (!put.ok) throw new Error(`Tải lên Cloudflare R2 thất bại (mã ${put.status}).`);
  return info.path as string;

export async function removeFiles(bucket: Bucket, paths: (string | null | undefined)[]) {
  const list = [...new Set(paths.filter((p): p is string => !!p))];
  if (!list.length) return;

  if (!MEDIA_URL) {
    const { error } = await supabase.storage.from(bucket).remove(list);
    if (error) console.warn(`Không xoá được tệp trong ${bucket}:`, error.message);
    return;
  }
  try {
    const res = await fetch("/api/media/delete", {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${await accessToken()}` },
      body: JSON.stringify({ keys: list.map((p) => `${bucket}/${p}`) }),
    });
    if (!res.ok) console.warn("Không xoá được tệp trên R2:", await res.text());
  } catch (e) {
    console.warn("Không xoá được tệp trên R2:", e);
  }
}

  if (!file.type.startsWith("image/")) throw new Error("Chỉ chấp nhận tệp ảnh.");
  if (file.size > 10 * 1024 * 1024) throw new Error("Ảnh tối đa 10 MB.");

export async function updateSong(song: Song, input: SongInput, userId: string, isAdmin: boolean): Promise<Song> {
  let audio_path = song.audio_path;
  let cover_path = song.cover_path;
  let newAudioPath: string | null = null;
  let newCoverPath: string | null = null;
  const oldAudio: string[] = [];
  const oldCover: string[] = [];

  try {
    if (input.newAudio) {
      newAudioPath = await uploadFile("music", input.newAudio, userId);
      audio_path = newAudioPath;
      oldAudio.push(song.audio_path);
    }
    if (input.newCover) {
      newCoverPath = await uploadFile("covers", input.newCover, userId);
      cover_path = newCoverPath;
      if (song.cover_path) oldCover.push(song.cover_path);
    } else if (input.removeCover && song.cover_path) {
      cover_path = null;
      oldCover.push(song.cover_path);
    }
  } catch (e) {
    await removeFiles("music", [newAudioPath]);
    throw e;
  }

  const { data, error } = await supabase.from("songs").update({
    title: input.title.trim(),
    artist: input.artist.trim() || null,
    album_id: input.albumId,
    track_no: input.trackNo,
    lyrics: input.lyrics.trim() || null,
    audio_path,
    cover_path,
    is_published: isAdmin ? song.is_published : false,
  }).eq("id", song.id).select().single();

  if (error) {
    await removeFiles("music", [newAudioPath]);
    await removeFiles("covers", [newCoverPath]);
    throw error;
  }
  await removeFiles("music", oldAudio);
  await removeFiles("covers", oldCover);

/** Địa chỉ công khai của Cloudflare R2, ví dụ https://media.phattaitam.com (để trống = dùng Supabase Storage) */
export const MEDIA_URL = (process.env.NEXT_PUBLIC_MEDIA_URL ?? "").replace(/\/+$/, "");

/**
 * URL công khai của tệp.
 * - download=true hoặc tên tệp: trình duyệt tải về thay vì mở.
 */
export function publicUrl(bucket: Bucket, path: string, download?: boolean | string): string {
  if (MEDIA_URL) {
    if (download) {
      const q = new URLSearchParams({ key: `${bucket}/${path}` });
      if (typeof download === "string") q.set("name", download);
      return `/api/media/download?${q.toString()}`;
    }
    return `${MEDIA_URL}/${bucket}/${encodePathClient(path)}`;
  }
  return supabase.storage.from(bucket).getPublicUrl(path, download ? { download } : undefined).data.publicUrl;
}
